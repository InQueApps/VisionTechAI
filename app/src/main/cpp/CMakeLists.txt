cmake_minimum_required(VERSION 3.18.1)
project("vision_segmenter")

# Native segmentation library (no OpenCV required; can be replaced with OpenCV later)
add_library(vision_segmenter SHARED segmenter_jni.cpp)
target_compile_options(vision_segmenter PRIVATE -Wall -Wextra -O2)
# 16 KB page alignment for Android 15+ / 16 KB page-size devices (required for arm64-v8a compatibility)
if(ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "x86_64")
  target_link_options(vision_segmenter PRIVATE "LINKER:-z,max-page-size=16384")
endif()
find_library(log-lib log)
target_link_libraries(vision_segmenter ${log-lib})

# Night vision: adaptive low-light enhancement
add_library(vision_nightvision SHARED nightvision_jni.cpp)
target_compile_options(vision_nightvision PRIVATE -Wall -Wextra -O2)
if(ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "x86_64")
  target_link_options(vision_nightvision PRIVATE "LINKER:-z,max-page-size=16384")
endif()
target_link_libraries(vision_nightvision ${log-lib})

# Depth postprocess (normalize + render to greyscale)
add_library(vision_depth SHARED depth_jni.cpp)
target_compile_options(vision_depth PRIVATE -Wall -Wextra -O2)
if(ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "x86_64")
  target_link_options(vision_depth PRIVATE "LINKER:-z,max-page-size=16384")
endif()
target_link_libraries(vision_depth ${log-lib})

# Text translation: cluster words into lines by location
add_library(vision_translator SHARED translation_jni.cpp)
target_compile_options(vision_translator PRIVATE -Wall -Wextra -O2)

endif()
target_link_libraries(vision_translator ${log-lib})

# Video: ARGB to I420 for processed recording
add_library(vision_video SHARED video_jni.cpp)
